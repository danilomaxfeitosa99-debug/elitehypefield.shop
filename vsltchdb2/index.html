    let count = 81; // Quantidade inicial de potes
    const counterElement = document.querySelector('.counter_left');
    const targetCount = 6; // Contagem alvo
    const totalTime = 17 * 60 * 1000; // 17 minutos em milissegundos
    const remainingTimeAfterFirstMinute = totalTime - 60 * 1000; // Tempo restante após o primeiro minuto
    const stepTimeAfterFirstMinute = 1000; // Tempo entre cada redução após o primeiro minuto (1 segundo)
    const totalStepsAfterFirstMinute = remainingTimeAfterFirstMinute / stepTimeAfterFirstMinute;
    const decrementPerStepAfterFirstMinute = (count - 54 - targetCount) / totalStepsAfterFirstMinute; // Redução suave
    let firstMinuteInterval;
    let counterInterval;
    function updateCounterFirstMinute() {
    count -= 1; // Reduz 1 a cada 2 segundos
    counterElement.innerText = count;
    // Para o intervalo após o primeiro minuto
    if (count <= 54) {
    clearInterval(firstMinuteInterval);
    startSmoothReduction(); // Inicia a redução suave
    }
    }
    function updateCounterSmooth() {
    count -= decrementPerStepAfterFirstMinute;
    // Impede valores negativos e garante que não passe do alvo
    if (count < targetCount) {
    count = targetCount;
    }
    // Atualiza o elemento na tela
    counterElement.innerText = Math.ceil(count);
    // Para a contagem quando atingir o valor alvo
    if (count <= targetCount) {
    clearInterval(counterInterval);
    }
    }
    function startSmoothReduction() {
    counterInterval = setInterval(updateCounterSmooth, stepTimeAfterFirstMinute);
    }
    function startCounter() {
    // Reduz 1 a cada 2 segundos durante o primeiro minuto
    firstMinuteInterval = setInterval(updateCounterFirstMinute, 2000);
    // Para todos os intervalos após 17 minutos
    setTimeout(() => {
    clearInterval(firstMinuteInterval);
    clearInterval(counterInterval);
    count = targetCount;
    counterElement.innerText = count;
    }, totalTime);
